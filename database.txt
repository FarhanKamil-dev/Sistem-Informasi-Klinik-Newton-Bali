CREATE DATABASE IF NOT EXISTS klinik_jakarta;
USE klinik_jakarta;

-- Membuat tabel admin
CREATE TABLE admin (
    id INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(50) NOT NULL,
    password VARCHAR(255) NOT NULL
);

-- Membuat tabel artikel dengan menambahkan kolom jam
CREATE TABLE artikel (
    id INT AUTO_INCREMENT PRIMARY KEY,
    judul VARCHAR(255) NOT NULL,
    isi TEXT NOT NULL,
    gambar VARCHAR(255),
    tanggal DATE NOT NULL,
    jam TIME NOT NULL -- Menambahkan kolom jam
);
